
#[cfg(test)]
mod routing_tests {
    use super::*;
    #[test]
    fn complete_group_structure_matches_original() {
        for n in [8192usize, 8193, 21006, 30089, 32767] {
            for kind in 0..7 {
                let unique=match kind {0=>0,1=>1,2=>17,3=>n*80/100,4=>n*80/100+1,_=>n};
                let words:Vec<_>=(0..n).map(|i|if unique==0||(kind==6&&i%7==0){[0;4]}else{[(i%unique+1)as u64,0,0,0]}).collect();
                let old=crate::msm_affine::grouping::scalar_groups(&words);
                let new=group_or_decline(&words);
                assert_eq!(new.is_none(),old.is_none(),"n={n},kind={kind}");
                if let (Some(a),Some(b))=(old,new){assert_eq!(a.len,b.len);assert_eq!(a.values,b.values);assert_eq!(a.order,b.order);assert_eq!(a.starts,b.starts);assert_eq!(a.counts,b.counts);}
            }
        }
    }
    #[test]
    fn selector_only_admits_the_existing_small_interval() {
        for n in [0usize,1,255,256,8191,32768,65536,211_000_000,usize::MAX] {assert!(!eligible(n));}
        for n in [8192usize,8193,21006,30089,32767] {assert!(eligible(n));}
    }
}
