    // BEGIN STREAMED_COMPLEX_B
    /// Complex product with three shared raw products and two REDCs.
    /// Inputs have normalized limbs and lie below 2p; outputs lie below 2p.
    /// The real CIOS accumulator is signed. Arithmetic carries preserve its
    /// exact integer value; adding p after REDC makes it nonnegative.
    #[cfg(zopro_gauss_positive)]
    #[inline(always)]
    pub fn complex_product_streamed(a0: &Self, a1: &Self, b0: &Self, b1: &Self) -> [Self; 2] {
        assert!(C::P[3] < (1u64 << 62));
        unsafe {
            let zero = Self::splat(0);
            let mask = Self::splat(MASK);
            let p = split(C::P).map(Self::splat);
            let inv = Self::splat(C::INV & MASK);
            // Exact sums, not residues: each is below 4p < 2^256.
            let mut a = [zero; 5];
            let mut ca = zero;
            for j in 0..5 {
                let av = _mm256_add_epi64(_mm256_add_epi64(a0.0[j], a1.0[j]), ca);
                ca = _mm256_srli_epi64::<52>(av);
                a[j] = _mm256_and_si256(av, mask);
            }
            let mut cb = zero;
            let mut x = [zero; 6];
            let mut y = [zero; 6];
            macro_rules! row {
                ($i:literal) => {{
                    // Stream the exact summed-B digit used by this row.
                    let bv = _mm256_add_epi64(_mm256_add_epi64(b0.0[$i], b1.0[$i]), cb);
                    cb = _mm256_srli_epi64::<52>(bv);
                    let bi = _mm256_and_si256(bv, mask);
                    macro_rules! product {
                        ($j:literal, $k:literal) => {{
                            let al = _mm256_madd52lo_epu64(zero, a0.0[$j], b0.0[$i]);
                            let ah = _mm256_madd52hi_epu64(zero, a0.0[$j], b0.0[$i]);
                            let bl = _mm256_madd52lo_epu64(zero, a1.0[$j], b1.0[$i]);
                            let bh = _mm256_madd52hi_epu64(zero, a1.0[$j], b1.0[$i]);
                            x[$j] = _mm256_add_epi64(x[$j], _mm256_sub_epi64(al, bl));
                            x[$k] = _mm256_add_epi64(x[$k], _mm256_sub_epi64(ah, bh));
                            y[$j] = _mm256_sub_epi64(
                                _mm256_madd52lo_epu64(y[$j], a[$j], bi),
                                _mm256_add_epi64(al, bl),
                            );
                            y[$k] = _mm256_sub_epi64(
                                _mm256_madd52hi_epu64(y[$k], a[$j], bi),
                                _mm256_add_epi64(ah, bh),
                            );
                        }};
                    }
                    product!(0, 1);
                    product!(1, 2);
                    product!(2, 3);
                    product!(3, 4);
                    product!(4, 5);
                    macro_rules! cancel {
                        ($t:ident) => {{
                            let m = _mm256_madd52lo_epu64(zero, $t[0], inv);
                            let m = if MIXED && $i == 4 {
                                _mm256_and_si256(m, Self::splat((1u64 << 48) - 1))
                            } else {
                                _mm256_and_si256(m, mask)
                            };
                            let low = if MIXED && $i == 4 {
                                let c = _mm256_srai_epi64::<48>(_mm256_add_epi64(
                                    $t[0], Self::splat((1u64 << 48) - 1),
                                ));
                                _mm256_madd52hi_epu64(c, _mm256_slli_epi64::<4>(m), p[0])
                            } else {
                                let c = _mm256_srai_epi64::<52>(_mm256_add_epi64($t[0], mask));
                                _mm256_madd52hi_epu64(c, m, p[0])
                            };
                            macro_rules! reduction {
                                ($j:literal, $k:literal) => {{
                                    $t[$j] = _mm256_madd52lo_epu64($t[$j], m, p[$j]);
                                    $t[$k] = _mm256_madd52hi_epu64($t[$k], m, p[$j]);
                                }};
                            }
                            reduction!(1, 2);
                            reduction!(2, 3);
                            reduction!(3, 4);
                            reduction!(4, 5);
                            if MIXED && $i == 4 {
                                let mask48 = Self::splat((1u64 << 48) - 1);
                                $t = [
                                    _mm256_add_epi64(low, _mm256_slli_epi64::<4>(_mm256_and_si256($t[1], mask48))),
                                    _mm256_add_epi64(_mm256_srai_epi64::<48>($t[1]), _mm256_slli_epi64::<4>(_mm256_and_si256($t[2], mask48))),
                                    _mm256_add_epi64(_mm256_srai_epi64::<48>($t[2]), _mm256_slli_epi64::<4>(_mm256_and_si256($t[3], mask48))),
                                    _mm256_add_epi64(_mm256_srai_epi64::<48>($t[3]), _mm256_slli_epi64::<4>(_mm256_and_si256($t[4], mask48))),
                                    _mm256_add_epi64(_mm256_srai_epi64::<48>($t[4]), _mm256_slli_epi64::<4>($t[5])),
                                    zero,
                                ];
                            } else {
                                $t[1] = _mm256_add_epi64($t[1], low);
                                $t = [$t[1], $t[2], $t[3], $t[4], $t[5], zero];
                            }
                        }};
                    }
                    cancel!(x);
                    cancel!(y);
                }};
            }
            row!(0);
            row!(1);
            row!(2);
            row!(3);
            row!(4);
            for j in 0..5 {
                x[j] = _mm256_add_epi64(x[j], p[j]);
            }
            for j in 0..4 {
                x[j + 1] = _mm256_add_epi64(x[j + 1], _mm256_srai_epi64::<52>(x[j]));
                x[j] = _mm256_and_si256(x[j], mask);
                y[j + 1] = _mm256_add_epi64(y[j + 1], _mm256_srai_epi64::<52>(y[j]));
                y[j] = _mm256_and_si256(y[j], mask);
            }
            [
                Self::reduce([x[0], x[1], x[2], x[3], x[4]], crate::portable::add4(C::P, C::P).0),
                Self::reduce([y[0], y[1], y[2], y[3], y[4]], crate::portable::add4(C::P, C::P).0),
            ]
        }
    }
    // END STREAMED_COMPLEX_B

    // BEGIN NONNEGATIVE_GAUSS
    /// Three nonnegative raw products with two simultaneous unsigned REDCs.
    /// S=a0+a1, N=2p-a1, T=b0+b1, D=b1+2p-b0 are exact integers.
    /// X=S*b0+N*T=real+2p*T, Y=S*b0+a0*D=imag+2p*a0.
    /// Both are <=12p². Prepared raw arrays can exceed the public 2p bound.
    #[cfg(zopro_gauss_positive)]
    #[inline(always)]
    pub fn complex_product_gauss(a0:&Self,a1:&Self,b0:&Self,b1:&Self)->[Self;2]{
        assert!(C::P[3] < (1u64 << 62));
        unsafe {
            let zero=Self::splat(0);
            let mask=Self::splat(MASK);
            let p=split(C::P).map(Self::splat);
            let two_p=split(crate::portable::add4(C::P,C::P).0).map(Self::splat);
            let inv=Self::splat(C::INV & MASK);
            let mut s=[zero;5];let mut n=[zero;5];
            let mut cs=zero;let mut cn=zero;
            for j in 0..5 {
                let sv=_mm256_add_epi64(_mm256_add_epi64(a0.0[j],a1.0[j]),cs);
                cs=_mm256_srli_epi64::<52>(sv);s[j]=_mm256_and_si256(sv,mask);
                let nv=_mm256_add_epi64(_mm256_sub_epi64(two_p[j],a1.0[j]),cn);
                cn=_mm256_srai_epi64::<52>(nv);n[j]=_mm256_and_si256(nv,mask);
            }
            let mut ct=zero;let mut cd=zero;
            let mut x=[zero;6];let mut y=[zero;6];
            macro_rules! row {
                ($i:literal)=>{{
                    let tv=_mm256_add_epi64(_mm256_add_epi64(b0.0[$i],b1.0[$i]),ct);
                    ct=_mm256_srli_epi64::<52>(tv);let ti=_mm256_and_si256(tv,mask);
                    let dv=_mm256_add_epi64(_mm256_sub_epi64(_mm256_add_epi64(b1.0[$i],two_p[$i]),b0.0[$i]),cd);
                    cd=_mm256_srai_epi64::<52>(dv);let di=_mm256_and_si256(dv,mask);
                    macro_rules! product {
                        ($j:literal,$k:literal)=>{{
                            let pl=_mm256_madd52lo_epu64(zero,s[$j],b0.0[$i]);
                            let ph=_mm256_madd52hi_epu64(zero,s[$j],b0.0[$i]);
                            x[$j]=_mm256_add_epi64(_mm256_madd52lo_epu64(x[$j],n[$j],ti),pl);
                            x[$k]=_mm256_add_epi64(_mm256_madd52hi_epu64(x[$k],n[$j],ti),ph);
                            y[$j]=_mm256_add_epi64(_mm256_madd52lo_epu64(y[$j],a0.0[$j],di),pl);
                            y[$k]=_mm256_add_epi64(_mm256_madd52hi_epu64(y[$k],a0.0[$j],di),ph);
                        }};
                    }
                    product!(0,1);product!(1,2);product!(2,3);product!(3,4);product!(4,5);
                    macro_rules! cancel {
                        ($t:ident) => {{
                            let m = _mm256_madd52lo_epu64(zero, $t[0], inv);
                            let m = if MIXED && $i == 4 {
                                _mm256_and_si256(m, Self::splat((1u64 << 48) - 1))
                            } else {
                                _mm256_and_si256(m, mask)
                            };
                            let low = if MIXED && $i == 4 {
                                let c = _mm256_srli_epi64::<48>(_mm256_add_epi64(
                                    $t[0], Self::splat((1u64 << 48) - 1),
                                ));
                                _mm256_madd52hi_epu64(c, _mm256_slli_epi64::<4>(m), p[0])
                            } else {
                                let c = _mm256_srli_epi64::<52>(_mm256_add_epi64($t[0], mask));
                                _mm256_madd52hi_epu64(c, m, p[0])
                            };
                            macro_rules! reduction {
                                ($j:literal, $k:literal) => {{
                                    $t[$j] = _mm256_madd52lo_epu64($t[$j], m, p[$j]);
                                    $t[$k] = _mm256_madd52hi_epu64($t[$k], m, p[$j]);
                                }};
                            }
                            reduction!(1, 2);
                            reduction!(2, 3);
                            reduction!(3, 4);
                            reduction!(4, 5);
                            if MIXED && $i == 4 {
                                let mask48 = Self::splat((1u64 << 48) - 1);
                                $t = [
                                    _mm256_add_epi64(low, _mm256_slli_epi64::<4>(_mm256_and_si256($t[1], mask48))),
                                    _mm256_add_epi64(_mm256_srli_epi64::<48>($t[1]), _mm256_slli_epi64::<4>(_mm256_and_si256($t[2], mask48))),
                                    _mm256_add_epi64(_mm256_srli_epi64::<48>($t[2]), _mm256_slli_epi64::<4>(_mm256_and_si256($t[3], mask48))),
                                    _mm256_add_epi64(_mm256_srli_epi64::<48>($t[3]), _mm256_slli_epi64::<4>(_mm256_and_si256($t[4], mask48))),
                                    _mm256_add_epi64(_mm256_srli_epi64::<48>($t[4]), _mm256_slli_epi64::<4>($t[5])),
                                    zero,
                                ];
                            } else {
                                $t[1] = _mm256_add_epi64($t[1], low);
                                $t = [$t[1], $t[2], $t[3], $t[4], $t[5], zero];
                            }
                        }};
                    }
                    cancel!(x);cancel!(y);
                }};
            }
            row!(0);row!(1);row!(2);row!(3);row!(4);
            for j in 0..4 {
                x[j+1]=_mm256_add_epi64(x[j+1],_mm256_srli_epi64::<52>(x[j]));x[j]=_mm256_and_si256(x[j],mask);
                y[j+1]=_mm256_add_epi64(y[j+1],_mm256_srli_epi64::<52>(y[j]));y[j]=_mm256_and_si256(y[j],mask);
            }
            let x=[x[0],x[1],x[2],x[3],x[4]];let y=[y[0],y[1],y[2],y[3],y[4]];
            if MIXED {[Self::reduce_fused(x),Self::reduce_fused(y)]}
            else {[Self(x,PhantomData),Self(y,PhantomData)]}
        }
    }
    // END NONNEGATIVE_GAUSS
